代码截图2：等距方位投影坐标变换

// 等距方位投影：将世界坐标转换为屏幕坐标 const worldToScreen = (wx, wy, acX, acY, heading, pxPerNM) => { const dx = wx - acX; // 目标点相对于飞机的X偏移 const dy = wy - acY; // 目标点相对于飞机的Y偏移 const rad = heading * Math.PI / 180; // 航向角度转弧度 // 航向旋转矩阵：将地理坐标旋转至航向朝上 const rx = dx * Math.cos(rad) + dy * Math.sin(rad); const ry = -dx * Math.sin(rad) + dy * Math.cos(rad); // 缩放并返回屏幕坐标（Y轴取反） return { sx: rx * pxPerNM, sy: -ry * pxPerNM }; };

class MapDataService { // 从URL加载地图数据 async loadMapFromUrl(url) { const response = await fetch(url); const rawData = await response.json(); return this.validateAndNormalizeMapData(rawData); } // 从本地文件加载地图数据 async loadMapFromFile(file) { const text = await file.text(); const rawData = JSON.parse(text); return this.validateAndNormalizeMapData(rawData); } // 验证并标准化地图数据 validateAndNormalizeMapData(rawData) { return { metadata: { name: rawData.metadata?.name || 'Unnamed Map', version: rawData.metadata?.version || '1.0', bounds: rawData.metadata?.bounds || null }, waypoints: this.normalizeWaypoints(rawData.waypoints || []), navaids: this.normalizeNavaids(rawData.navaids || []), airways: rawData.airways || [], terrain: rawData.terrain || [], defaultRoutes: rawData.defaultRoutes || [], }; } }
